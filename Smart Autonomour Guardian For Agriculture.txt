#include<SoftwareSerial.h> 
#define IN1 5 
#define IN2 4 
#define IN3 6 
#define IN4 7 
#define DHT_PIN 8 
#define DHT_TYPE DHT11 
#include <DHT.h> 
int ir=12; 
// DHT setup 
DHT dht(DHT_PIN, DHT_TYPE); 
int pump = 2; 
SoftwareSerial Serialp(10,11); 
int pumpip=9; 
bool pumpset=false; 
bool forwardset=false; 
int pwm=3; 
unsigned long pt=0; 
float temperature = 0; 
float humidity = 0; 
void setup() { 
// put your setup code here, to run once: 
dht.begin(); 
Serialp.begin(9600); 
Serial.begin(9600); 
pinMode(IN1, OUTPUT); 
pinMode(IN2, OUTPUT); 
pinMode(IN3, OUTPUT); 
pinMode(IN4, OUTPUT); 
pinMode(pumpip,INPUT); 
pinMode(pump,OUTPUT); 
pinMode(pwm,OUTPUT); 
digitalWrite(pump,HIGH); 
} 
void loop() { 
if(millis()-pt>1000){ 
// Read temperature and humidity 
temperature = dht.readTemperature(); 
humidity = dht.readHumidity(); 
Serialp.print(" | Temp: "); 
Serialp.print(temperature); 
Serialp.print("C | Hum: "); 
Serialp.print(humidity); 
Serialp.println("%"); 
if(!digitalRead(ir)){ 
Serialp.println("Warning! obstical alert "); 
} 
pt=millis(); 
} 
if(!digitalRead(pumpip) && pumpset && forwardset){ 
delay(400); 
PUMPON(); 
delay(1000); 
PUMPOFF(); 
}else{ 
PUMPOFF(); 
} 
if(Serialp.available()>0){ 
String motiondata=Serialp.readString(); 
int direction = motiondata.toInt(); 
if(direction==1){ 
carturnright(); 
forwardset=true; 
} 
if(direction==2){ 
carturnleft(); 
pumpset=false; 
forwardset=false; 
} 
if(direction==3){ 
carforward(); 
pumpset=false; 
forwardset=false; 
} 
if(direction==4){ 
carbackward(); 
pumpset=false; 
forwardset=false; 
} 
if(direction==5){ 
carStop(); 
pumpset=false; 
forwardset=false; 
} 
if(direction==6){ 
pumpset=true; 
} 
} 
} 
void carforward() { 
analogWrite(pwm,255); 
digitalWrite(IN1, LOW); 
digitalWrite(IN2, HIGH); 
digitalWrite(IN3, LOW); 
digitalWrite(IN4, HIGH); 
} 
void carbackward() { 
analogWrite(pwm,255); 
digitalWrite(IN1, HIGH); 
digitalWrite(IN2, LOW); 
digitalWrite(IN3, HIGH); 
digitalWrite(IN4, LOW); 
} 
void carturnleft() { 
analogWrite(pwm,255); 
digitalWrite(IN1, LOW); 
digitalWrite(IN2, HIGH); 
digitalWrite(IN3, HIGH); 
digitalWrite(IN4, LOW); 
} 
void carturnright() { 
analogWrite(pwm,140); 
digitalWrite(IN1, HIGH); 
digitalWrite(IN2, LOW); 
digitalWrite(IN3, LOW); 
digitalWrite(IN4, HIGH); 
} 
void carStop() { 
digitalWrite(IN1, LOW); 
digitalWrite(IN2, LOW); 
digitalWrite(IN3, LOW); 
digitalWrite(IN4, LOW); 
} 
void PUMPON(){ 
digitalWrite(pump,LOW); 
} 
void PUMPOFF(){ 
digitalWrite(pump,HIGH);

import cv2 
import time 
import threading 
import torch 
import requests 
import numpy as np 
from ultralytics import YOLO 
from ultralytics.utils.plotting import Annotator 
# Configuration 
ESP32_URL = "http://192.168.4.1" # ESP32-CAM hotspot address 
STREAM_URL = f"{ESP32_URL}/stream" 
MODEL_PATH = 'best.pt' # Your YOLOv11 model path 
IMG_SIZE = 320 # Inference size 
CONF_THRES = 0.5 # Confidence threshold 
target="plant_b" 
# Global variables 
frame = None 
lock = threading.Lock()

start_time = time.time() 
# Initialize YOLOv11 model 
device = 'cuda' if torch.cuda.is_available() else 'cpu' 
model = YOLO(MODEL_PATH) 
model.to(device) 
names = model.name 
# Initialize video capture 
cap = cv2.VideoCapture(STREAM_URL) 
def control_esp32_led(state): 
"""Control ESP32-CAM LED""" 
try: 
response = requests.get(f"{ESP32_URL}/led?state={state}", timeout=1) 
print(f"LED {state}: {response.text}") 
except Exception as e: 
print(f"Error controlling LED: {e}") 
def capture_frames(): 
"""Thread for capturing video frames""" 
global frame, cap 
while True: 
try: 
if cap is None or not cap.isOpened(): 
cap = cv2.VideoCapture(STREAM_URL) 
time.sleep(2) 
continue 
ret, captured_frame = cap.read() 
if not ret: 
print("Frame read failed, reconnecting...") 
cap.release() 
cap = None 
continue 
with lock: 
frame = captured_frame 
except Exception as e: 
print(f"Error in capture thread: {e}") 
if cap: 
cap.release() 
cap = None 
time.sleep(1) 
# Start frame capture thread 
capture_thread = threading.Thread(target=capture_frames, daemon=True) 
capture_thread.start() 
# Main processing loop 
while True: 
with lock: 
if frame is None: 
continue 
# Resize frame for display 
display_frame = cv2.resize(frame, (640, 480)) 
# Use original frame for processing (YOLOv11 handles resizing internally) 
process_frame = frame.copy() 
# Perform inference 
results = model(process_frame, imgsz=IMG_SIZE, conf=CONF_THRES, verbose=False) 
# Process detections 
current_detection = False

# Check if there are any detections 
if result.boxes is not None and len(result.boxes) > 0: 
# Create annotator for drawing

# Extract boxes, classes, and confidence scores

confidences = result.boxes.conf.cpu().numpy() 
# Draw bounding boxes and labels 
for box, cls, conf in zip(boxes, classes, confidences): 
# Scale boxes to display frame size if needed

box[3] *= scale_y
